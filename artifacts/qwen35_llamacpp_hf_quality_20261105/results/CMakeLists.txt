cmake_minimum_required(VERSION 3.18)
project(infer_opt_llama_quality LANGUAGES C CXX)
if(NOT LLAMA_CPP_SOURCE)
    message(FATAL_ERROR "Set LLAMA_CPP_SOURCE to a pinned llama.cpp checkout")
endif()
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_CPP_SOURCE}" llama EXCLUDE_FROM_ALL)
add_executable(infer-opt-llama-replay replay.cpp)
target_link_libraries(infer-opt-llama-replay PRIVATE llama)
target_compile_features(infer-opt-llama-replay PRIVATE cxx_std_17)
